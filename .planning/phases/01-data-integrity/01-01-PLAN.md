<objective>
Rewrite the POST /Reservations handler to wrap all DB mutations in a single atomic transaction and prevent overbooking via a SELECT ... FOR UPDATE row lock on the room row before the availability check runs.

Purpose: A crash between any two current pool.query() calls in the POST handler leaves an orphan reservation row or missing guest rows. Two concurrent requests for the same room can both pass the availability check before either inserts — overbooking is currently possible via race condition.

Output: The POST /Reservations handler in reservations.routes.ts uses pool.connect() + BEGIN/COMMIT/ROLLBACK and locks the room row with SELECT ... FOR UPDATE before checking availability. All inner queries use client.query(), not the module-level query() helper.
</objective>

   const client = await pool.connect();
   try {
     await client.query("BEGIN");

     // Step 1: Lock the room row — serializes concurrent reservation attempts for this room.
     // FOR UPDATE blocks any other transaction that tries to lock the same room row until COMMIT/ROLLBACK.
     const roomRows = await client.query<RoomForReservationRow>(
       `SELECT id, daily_price::text AS daily_price, status
        FROM rooms WHERE id = $1 FOR UPDATE`,
       [req.body.roomId]
     );
     const room = roomRows.rows[0];
     if (!room) throw new HttpError(404, "Quarto nao encontrado.");
     if (isMaintenanceRoomStatus(room.status))
       throw new HttpError(400, "Quarto em manutencao nao pode receber reservas.");

     // Step 2: Validate client exists (inside transaction so the read is consistent).
     const clientRows = await client.query<ClientReferenceRow>(
       `SELECT id FROM clients WHERE id = $1 LIMIT 1`,
       [req.body.clientId]
     );
     if ((clientRows.rowCount ?? 0) === 0)
       throw new HttpError(404, "Cliente nao encontrado.");

     // Step 3: Validate dates.
     validateReservationDates(req.body.checkInDate, req.body.checkOutDate);

     // Step 4: Availability check — safe because the room row is locked since Step 1.
     // Using client.query() so the check runs in the same transaction as the insert.
     const conflicts = await client.query<{ id: string }>(
       `SELECT id FROM reservations
        WHERE room_id = $1
          AND status = ANY($5::text[])
          AND ($4::uuid IS NULL OR id <> $4::uuid)
          AND check_in_date < $3::timestamptz
          AND check_out_date > $2::timestamptz
        LIMIT 1`,
       [req.body.roomId, req.body.checkInDate, req.body.checkOutDate, null, ACTIVE_RESERVATION_STATUSES]
     );
     if ((conflicts.rowCount ?? 0) > 0)
       throw new HttpError(409, "Ja existe uma reserva nesse quarto para o periodo informado.");

     // Step 5: Fetch pricing rules inside the transaction (consistent read).
     const priceRuleRows = await client.query<PricingRuleRow>(
       `SELECT id, name, description, min_age, max_age, price::text AS price FROM pricing_rules`
     );
     const pricingRules = mapPricingRules(priceRuleRows.rows);

     // Step 6: Build guest objects and calculate total.
     const guests: ReservationGuest[] = req.body.guests.map((g: ReservationGuest) => ({
       id: randomUUID(),
       reservationId: "",
       name: g.name,
       age: g.age,
       pricingRuleId: g.pricingRuleId ?? null
     }));
     const totalPrice = calculateReservationTotal(
       Number(room.daily_price),
       req.body.checkInDate,
       req.body.checkOutDate,
       guests,
       pricingRules
     );

     // Step 7: Insert the reservation row.
     const reservationId = randomUUID();
     await client.query(
       `INSERT INTO reservations (id, room_id, client_id, check_in_date, check_out_date, status, total_price)
        VALUES ($1, $2, $3, $4, $5, $6, $7)`,
       [reservationId, req.body.roomId, req.body.clientId,
        req.body.checkInDate, req.body.checkOutDate, req.body.status, totalPrice]
     );

     // Step 8: Insert guest rows. If any insert fails, ROLLBACK removes the reservation row too.
     for (const guest of guests) {
       await client.query(
         `INSERT INTO reservation_guests (id, reservation_id, name, age, pricing_rule_id)
          VALUES ($1, $2, $3, $4, $5)`,
         [guest.id, reservationId, guest.name, guest.age, guest.pricingRuleId]
       );
     }

     await client.query("COMMIT");

     // Fetch created reservation outside the transaction (read-only, no atomicity needed).
     const reservations = await getReservationsByIds([reservationId]);
     res.status(201).json(reservations[0]);
   } catch (error) {
     await client.query("ROLLBACK");
     throw error;
   } finally {
     client.release();
   }
